cmake_minimum_required(VERSION 3.20)
project(brotli_artifact_qa C)
add_executable(brotli-probe brotli-probe.c)
target_include_directories(brotli-probe PRIVATE "${BROTLI_ROOT}/include")
target_link_libraries(brotli-probe PRIVATE "${BROTLI_ROOT}/lib/brotlienc.lib" "${BROTLI_ROOT}/lib/brotlidec.lib" "${BROTLI_ROOT}/lib/brotlicommon.lib")
if(JXL_ROOT)
foreach(kind shared static)
  add_executable(jxl-${kind}-probe jxl-probe.c)
  target_include_directories(jxl-${kind}-probe PRIVATE "${JXL_ROOT}/include")
  if(kind STREQUAL "shared")
    target_link_libraries(jxl-${kind}-probe PRIVATE "${JXL_ROOT}/lib/jxl.lib" "${JXL_ROOT}/lib/jxl_cms.lib")
  else()
    target_compile_definitions(jxl-${kind}-probe PRIVATE JXL_STATIC_DEFINE JXL_CMS_STATIC_DEFINE)
    target_link_libraries(jxl-${kind}-probe PRIVATE "${JXL_ROOT}/lib/jxl_a.lib" "${JXL_ROOT}/lib/jxl_cms_a.lib" "${JXL_ROOT}/lib/hwy_a.lib" "${BROTLI_ROOT}/lib/brotlienc.lib" "${BROTLI_ROOT}/lib/brotlidec.lib" "${BROTLI_ROOT}/lib/brotlicommon.lib")
  endif()
endforeach()
endif()
